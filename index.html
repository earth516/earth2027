import fitz # PyMuPDF import re import os import time import threading import webbrowser import tkinter as tk from tkinter import filedialog, messagebox from PIL import Image, ImageChops from google.auth.transport.requests import Request from google.oauth2.credentials import Credentials from google_auth_oauthlib.flow import InstalledAppFlow from googleapiclient.discovery import build from googleapiclient.http import MediaFileUpload from googleapiclient.errors import HttpError SCOPES = [ 'https://www.googleapis.com/auth/documents', 'https://www.googleapis.com/auth/drive' ] # --------------------------------------------------------- # 0. 이미지 상하단 흰색 여백 자동 제거 함수 # --------------------------------------------------------- def trim_white_borders(image): bg = Image.new(image.mode, image.size, (255, 255, 255)) diff = ImageChops.difference(image, bg) diff = ImageChops.add(diff, diff, 2.0, -100) bbox = diff.getbbox() if bbox: return image.crop((0, max(0, bbox[1] - 5), image.width, min(image.height, bbox[3] + 5))) return image # --------------------------------------------------------- # 1. 캡처 보조 함수 (단일 / 분할 영역 결합) # --------------------------------------------------------- def capture_range(doc, start_info, end_info, out_path): p_start, y_s = start_info["page"], max(0, start_info["y"] - 5) p_end, y_e = end_info["page"], end_info["y"] - 5 if p_start == p_end: page = doc[p_start] rect = fitz.Rect(0, y_s, page.rect.width, max(y_s + 20, y_e)) pix = page.get_pixmap(clip=rect, dpi=130) pix.save(out_path) im = Image.open(out_path) trimmed = trim_white_borders(im) trimmed.save(out_path) elif p_end == p_start + 1: p1 = doc[p_start] p2 = doc[p_end] pix1 = p1.get_pixmap(clip=fitz.Rect(0, y_s, p1.rect.width, p1.rect.height), dpi=130) pix2 = p2.get_pixmap(clip=fitz.Rect(0, 0, p2.rect.width, max(20, y_e)), dpi=130) temp1 = out_path + "_p1.png" temp2 = out_path + "_p2.png" pix1.save(temp1) pix2.save(temp2) im1 = Image.open(temp1) im2 = Image.open(temp2) im1_trimmed = trim_white_borders(im1) im2_trimmed = trim_white_borders(im2) gap = 15 new_w = max(im1_trimmed.width, im2_trimmed.width) new_h = im1_trimmed.height + im2_trimmed.height + gap merged = Image.new('RGB', (new_w, new_h), (255, 255, 255)) merged.paste(im1_trimmed, (0, 0)) merged.paste(im2_trimmed, (0, im1_trimmed.height + gap)) merged.save(out_path) if os.path.exists(temp1): os.remove(temp1) if os.path.exists(temp2): os.remove(temp2) # --------------------------------------------------------- # 2. PDF 전체 좌표 분석 (양식 자동 감지) # --------------------------------------------------------- def process_pdf(pdf_path): doc = fitz.open(pdf_path) base_folder = os.path.dirname(os.path.abspath(__file__)) temp_folder = os.path.join(base_folder, "temp_images") if not os.path.exists(temp_folder): os.makedirs(temp_folder) q_points = [] code_points = [] a_points = [] h_points = [] choice_points = [] full_text = "" choice_targets = ["①", "1", "⑴"] for page_num in range(len(doc)): page = doc[page_num] full_text += page.get_text() + "\n" for r in page.search_for("[문제]"): q_points.append({"page": page_num, "y": r.y0}) for r in page.search_for("#문항코드"): # 표 헤더 하단(y1)에 여유를 두어 시작점으로 잡음 code_points.append({"page": page_num, "y": r.y1 + 15}) for r in page.search_for("[정답/모범답안]"): a_points.append({"page": page_num, "y": r.y0}) for r in page.search_for("#강"): h_points.append({"page": page_num, "y": r.y0}) for ct in choice_targets: for r in page.search_for(ct): choice_points.append({"page": page_num, "y": r.y0}) # [문제] 태그가 없으면 #문항코드 위치를 문제 시작점으로 대체 use_code_as_start = len(q_points) < len(a_points) and len(code_points) >= len(a_points) start_points = code_points if use_code_as_start else q_points # 해설 텍스트 파싱 ans_exp_pattern = r"\[정답/모범답안\](.*?)\[해설\](.*?)(?=\[정답/모범답안\]|#강|$)" text_matches = re.findall(ans_exp_pattern, full_text, re.DOTALL) total_probs = min(len(start_points), len(a_points), len(text_matches)) problems_data = [] for idx in range(total_probs): q = start_points[idx] a = a_points[idx] ans_text, exp_text = text_matches[idx] # 해설 종료 지점 (#강 표 직전) h_candidates = [ h for h in h_points if (h["page"] > a["page"]) or (h["page"] == a["page"] and h["y"] > a["y"] + 10) ] if h_candidates: exp_end = h_candidates[0] elif idx + 1 < len(start_points): exp_end = start_points[idx + 1] else: exp_end = {"page": len(doc) - 1, "y": doc[len(doc) - 1].rect.height} # <보기> 조건문 판단 # 문제 영역의 페이지 텍스트를 일부 발췌하여 검사 prob_page_text = doc[q["page"]].get_text() if a["page"] > q["page"]: prob_page_text += doc[a["page"]].get_text() is_bogi_type = ("<보기>" in prob_page_text or "보기" in prob_page_text) and ("있는 대로 고른 것은" in prob_page_text) prob_nobogi_end = a if is_bogi_type: c_candidates = [ c for c in choice_points if ((c["page"] > q["page"]) or (c["page"] == q["page"] and c["y"] > q["y"] + 15)) and ((c["page"] < a["page"]) or (c["page"] == a["page"] and c["y"] < a["y"] - 10)) ] if c_candidates: c_candidates.sort(key=lambda item: (item["page"], item["y"])) prob_nobogi_end = c_candidates[0] prob_img = os.path.join(temp_folder, f"prob_{idx+1}.png") prob_nobogi_img = os.path.join(temp_folder, f"prob_nobogi_{idx+1}.png") ans_img = os.path.join(temp_folder, f"ans_{idx+1}.png") both_img = os.path.join(temp_folder, f"both_{idx+1}.png") try: capture_range(doc, q, a, prob_img) capture_range(doc, q, prob_nobogi_end, prob_nobogi_img) capture_range(doc, a, exp_end, ans_img) capture_range(doc, q, exp_end, both_img) problems_data.append({ "index": idx + 1, "prob_img": prob_img, "prob_nobogi_img": prob_nobogi_img, "ans_img": ans_img, "both_img": both_img, "answer": ans_text.strip(), "explanation": exp_text.strip() }) except Exception: continue return problems_data # --------------------------------------------------------- # 3. 구글 드라이브 & 문서 API 통신 # --------------------------------------------------------- def create_drive_folder(drive_service, folder_name): folder_metadata = {'name': folder_name, 'mimeType': 'application/vnd.google-apps.folder'} folder = drive_service.files().create(body=folder_metadata, fields='id').execute() return folder.get('id') def safe_drive_upload(drive_service, file_path, folder_id, max_retries=4): file_metadata = {'name': os.path.basename(file_path), 'parents': [folder_id]} media = MediaFileUpload(file_path, mimetype='image/png') for attempt in range(max_retries): try: uploaded_file = drive_service.files().create(body=file_metadata, media_body=media, fields='id').execute() file_id = uploaded_file.get('id') drive_service.permissions().create(fileId=file_id, body={'type': 'anyone', 'role': 'reader'}).execute() time.sleep(0.3) return f"https://drive.google.com/uc?id={file_id}" except HttpError as err: if err.resp.status in [500, 502, 503, 504]: time.sleep(2 * (attempt + 1)) else: raise err except Exception: time.sleep(2) raise Exception(f"{os.path.basename(file_path)} 업로드 실패") def write_to_google_docs(problems_data, credentials_json, status_label, doc_mode): creds = None token_path = os.path.join(os.path.dirname(os.path.abspath(__file__)), 'token.json') if os.path.exists(token_path): creds = Credentials.from_authorized_user_file(token_path, SCOPES) if not creds or not creds.valid: if creds and creds.expired and creds.refresh_token: creds.refresh(Request()) else: flow = InstalledAppFlow.from_client_secrets_file(credentials_json, SCOPES) creds = flow.run_local_server(port=0) with open(token_path, 'w') as token: token.write(creds.to_json()) drive_service = build('drive', 'v3', credentials=creds) docs_service = build('docs', 'v1', credentials=creds) if doc_mode == "prob_only": title_name, folder_name = "EBS 과학 [문제만]", "EBS_문제만_DOCS" elif doc_mode == "ans_only": title_name, folder_name = "EBS 과학 [해설만]", "EBS_해설만_DOCS" else: title_name, folder_name = "EBS 과학 [문제+해설]", "EBS_통합_DOCS" status_label.config(text="드라이브 폴더 생성 중...") folder_id = create_drive_folder(drive_service, folder_name=folder_name) doc = docs_service.documents().create(body={'title': title_name}).execute() document_id = doc.get('documentId') drive_service.files().update(fileId=document_id, addParents=folder_id, fields='id, parents').execute() requests = [] total = len(problems_data) if doc_mode == "ans_only": ans_exp_full_text = "========================================\n [ 정답 및 해설 ]\n========================================\n\n" for p_data in problems_data: ans_exp_full_text += ( f"[{p_data['index']}번 문항]\n" f"▶ 정답: {p_data['answer']}\n" f"▶ 해설:\n{p_data['explanation']}\n" f"----------------------------------------\n\n" ) requests.append({'insertText': {'location': {'index': 1}, 'text': ans_exp_full_text}}) elif doc_mode == "both": ans_exp_full_text = "========================================\n [ 정답 및 해설편 ]\n========================================\n\n" for p_data in problems_data: ans_exp_full_text += ( f"[{p_data['index']}번 문항]\n" f"▶ 정답: {p_data['answer']}\n" f"▶ 해설:\n{p_data['explanation']}\n" f"----------------------------------------\n\n" ) requests.append({'insertText': {'location': {'index': 1}, 'text': ans_exp_full_text}}) requests.append({'insertPageBreak': {'location': {'index': 1}}}) for i, p_data in enumerate(reversed(problems_data)): problem_num = total - i status_label.config(text=f"문제 업로드 중... ({i+1}/{total})") img_url = safe_drive_upload(drive_service, p_data["prob_img"], folder_id=folder_id) if i < total - 1: requests.append({'insertPageBreak': {'location': {'index': 1}}}) requests.append({ 'insertInlineImage': { 'location': {'index': 1}, 'uri': img_url, 'objectSize': {'width': {'magnitude': 450, 'unit': 'PT'}} } }) requests.append({'insertText': {'location': {'index': 1}, 'text': f"[{problem_num}번 문항]\n\n"}}) elif doc_mode == "prob_only": for i, p_data in enumerate(reversed(problems_data)): problem_num = total - i status_label.config(text=f"문제 업로드 중... ({i+1}/{total})") img_url = safe_drive_upload(drive_service, p_data["prob_img"], folder_id=folder_id) if i < total - 1: requests.append({'insertPageBreak': {'location': {'index': 1}}}) requests.append({ 'insertInlineImage': { 'location': {'index': 1}, 'uri': img_url, 'objectSize': {'width': {'magnitude': 450, 'unit': 'PT'}} } }) requests.append({'insertText': {'location': {'index': 1}, 'text': f"[{problem_num}번 문항]\n\n"}}) status_label.config(text="문서에 서식을 적용하는 중입니다...") docs_service.documents().batchUpdate(documentId=document_id, body={'requests': requests}).execute() return document_id # --------------------------------------------------------- # 4. 로컬 이미지 폴더 저장 # --------------------------------------------------------- def save_images_to_folder(problems_data, pdf_path, img_mode, status_label): base_name = os.path.splitext(os.path.basename(pdf_path))[0] folder_map = { "prob_img": f"{base_name}_문제만_그림", "prob_nobogi": f"{base_name}_문제(선택지제외)_그림", "ans_img": f"{base_name}_해설만_그림", "both_img": f"{base_name}_문제해설_그림" } save_folder = os.path.join(os.path.dirname(pdf_path), folder_map[img_mode]) if not os.path.exists(save_folder): os.makedirs(save_folder) for p in problems_data: idx = p["index"] if img_mode == "prob_img": src = p["prob_img"] dst = os.path.join(save_folder, f"{idx}번_문제.png") elif img_mode == "prob_nobogi": src = p["prob_nobogi_img"] dst = os.path.join(save_folder, f"{idx}번_문제(보기제외).png") elif img_mode == "ans_img": src = p["ans_img"] dst = os.path.join(save_folder, f"{idx}번_해설.png") else: src = p["both_img"] dst = os.path.join(save_folder, f"{idx}번_문제및해설.png") Image.open(src).save(dst) return save_folder # --------------------------------------------------------- # 5. 작업 스레드 # --------------------------------------------------------- def run_task(pdf_path, action_type, status_label, buttons): try: for b in buttons: b.config(state="disabled") status_label.config(text="PDF 양식 분석 및 캡처 중...") problems_data = process_pdf(pdf_path) if not problems_data: messagebox.showwarning("결과", "문항을 감지하지 못했습니다.") status_label.config(text="대기 중...") return if action_type in ["prob_only", "ans_only", "both"]: cred_file = os.path.join(os.path.dirname(os.path.abspath(__file__)), 'credentials.json') doc_id = write_to_google_docs(problems_data, cred_file, status_label, action_type) status_label.config(text="작업 완료!") messagebox.showinfo("성공", "구글 문서 생성이 완료되었습니다!") webbrowser.open(f"https://docs.google.com/document/d/{doc_id}") else: status_label.config(text="로컬 이미지 정리 중...") out_folder = save_images_to_folder(problems_data, pdf_path, action_type, status_label) status_label.config(text="저장 완료!") messagebox.showinfo("성공", f"이미지가 성공적으로 저장되었습니다!\n경로: {out_folder}") os.startfile(out_folder) except Exception as e: messagebox.showerror("에러", f"오류 발생:\n{str(e)}") status_label.config(text="오류 발생") finally: for b in buttons: b.config(state="normal") def handle_click(action_type, status_label, buttons): file_path = filedialog.askopenfilename( title="PDF 파일을 선택하세요", filetypes=[("PDF Files", "*.pdf")] ) if file_path: threading.Thread(target=run_task, args=(file_path, action_type, status_label, buttons)).start() # --------------------------------------------------------- # 6. UI 화면 # --------------------------------------------------------- root = tk.Tk() root.title("EBS 과학 통합 추출기") root.geometry("540x390") root.eval('tk::PlaceWindow . center') tk.Label(root, text="원하는 출력 형식을 선택하세요", font=("Helvetica", 14, "bold")).pack(pady=10) status_lbl = tk.Label(root, text="대기 중... 원하는 작업 버튼을 클릭하세요.", fg="blue") status_lbl.pack(pady=3) buttons = [] frame_docs = tk.LabelFrame(root, text=" Google Docs 문서 자동 생성 ", font=("Helvetica", 10, "bold"), padx=10, pady=6) frame_docs.pack(fill="x", padx=15, pady=4) btn_d1 = tk.Button(frame_docs, text="📄 문제만 doc", width=14, height=2, command=lambda: handle_click("prob_only", status_lbl, buttons)) btn_d1.grid(row=0, column=0, padx=4) btn_d2 = tk.Button(frame_docs, text="📝 해설만 doc", width=14, height=2, command=lambda: handle_click("ans_only", status_lbl, buttons)) btn_d2.grid(row=0, column=1, padx=4) btn_d3 = tk.Button(frame_docs, text="📚 문제+해설 doc", width=16, height=2, bg="#e8f0fe", command=lambda: handle_click("both", status_lbl, buttons)) btn_d3.grid(row=0, column=2, padx=4) frame_imgs = tk.LabelFrame(root, text=" 컴퓨터 폴더에 그림(PNG)으로 저장 ", font=("Helvetica", 10, "bold"), padx=10, pady=6) frame_imgs.pack(fill="x", padx=15, pady=4) btn_i1 = tk.Button(frame_imgs, text="🖼️ 문제만 그림", width=14, height=2, command=lambda: handle_click("prob_img", status_lbl, buttons)) btn_i1.grid(row=0, column=0, padx=4, pady=3) btn_i_nobogi = tk.Button(frame_imgs, text="✂️ 문제(선택지제외)", width=14, height=2, bg="#e6ffed", command=lambda: handle_click("prob_nobogi", status_lbl, buttons)) btn_i_nobogi.grid(row=0, column=1, padx=4, pady=3) btn_i2 = tk.Button(frame_imgs, text="🖼️ 해설만 그림", width=16, height=2, command=lambda: handle_click("ans_img", status_lbl, buttons)) btn_i2.grid(row=0, column=2, padx=4, pady=3) btn_i3 = tk.Button(frame_imgs, text="🖼️ 문제+해설 그림", width=48, height=2, bg="#fef7e0", command=lambda: handle_click("both_img", status_lbl, buttons)) btn_i3.grid(row=1, column=0, columnspan=3, padx=4, pady=3) buttons.extend([btn_d1, btn_d2, btn_d3, btn_i1, btn_i_nobogi, btn_i2, btn_i3]) root.mainloop()
